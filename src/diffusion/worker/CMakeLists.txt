cmake_minimum_required(VERSION 3.25)
project(VulkanaDiffusionWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VULKANA_DIFFUSION_RELEASE "Build a portable CUDA worker for distribution" OFF)
if(VULKANA_DIFFUSION_RELEASE)
    if(NOT SD_CUDA OR SD_VULKAN)
        message(FATAL_ERROR "Release worker bundles currently support the CUDA backend only")
    endif()
    if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
        message(FATAL_ERROR "Release worker bundles require CMAKE_BUILD_TYPE=Release")
    endif()
    # Keep the CPU fallback usable on AVX2 machines, independently of the
    # build host. GPU kernels include native RTX 30/40/50 targets and generic
    # Hopper PTX for forward JIT compatibility (120a PTX is architecture-specific).
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    foreach(feature AVX AVX2 F16C FMA BMI2)
        set(GGML_${feature} ON CACHE BOOL "" FORCE)
    endforeach()
    foreach(feature CPU_ALL_VARIANTS AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 AVX_VNNI AMX_TILE AMX_INT8 AMX_BF16)
        set(GGML_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP_FETCH OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
    set(GGML_CUDA_CCCL_VERSION "" CACHE STRING "Use the selected CUDA SDK's CCCL headers" FORCE)
    set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES "86-real;89-real;120a-real;90-virtual" CACHE STRING "Release GPU targets")
    endif()
    if(NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "86-real;89-real;120a-real;90-virtual"
       AND NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "86-real;89-real;120-real;90-virtual")
        message(FATAL_ERROR "Release CUDA workers require native 86/89/120(a) and generic 90-virtual PTX targets")
    endif()
endif()
if(MSVC)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
set(VULKANA_DIFFUSION_ENGINE_DIR "" CACHE PATH "Existing stable-diffusion.cpp checkout")
if(NOT EXISTS "${VULKANA_DIFFUSION_ENGINE_DIR}/include/stable-diffusion.h")
    message(FATAL_ERROR "Set VULKANA_DIFFUSION_ENGINE_DIR to the pinned engine checkout; this build downloads nothing")
endif()
find_package(Git REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine-lock.json" engine_lock)
string(JSON engine_revision GET "${engine_lock}" engine revision)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${VULKANA_DIFFUSION_ENGINE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE actual_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual_revision STREQUAL engine_revision)
    message(FATAL_ERROR "Engine revision differs from engine-lock.json; review and record the new baseline first")
endif()
string(JSON dependency_count LENGTH "${engine_lock}" dependencies)
math(EXPR dependency_last "${dependency_count} - 1")
foreach(index RANGE 0 ${dependency_last})
    string(JSON dependency MEMBER "${engine_lock}" dependencies ${index})
    string(JSON expected GET "${engine_lock}" dependencies "${dependency}")
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${VULKANA_DIFFUSION_ENGINE_DIR}/${dependency}" rev-parse HEAD
        OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Dependency ${dependency} differs from engine-lock.json")
    endif()
endforeach()
# Verify the recorded downstream patches without modifying the source checkout.
string(JSON patch_count LENGTH "${engine_lock}" patches)
if(patch_count GREATER 0)
    math(EXPR patch_last "${patch_count} - 1")
    foreach(index RANGE 0 ${patch_last})
        string(JSON patch_path GET "${engine_lock}" patches ${index} path)
        string(JSON patch_sha256 GET "${engine_lock}" patches ${index} sha256)
        set(patch_file "${CMAKE_CURRENT_SOURCE_DIR}/../${patch_path}")
        file(SHA256 "${patch_file}" actual_sha256)
        if(NOT actual_sha256 STREQUAL patch_sha256)
            message(FATAL_ERROR "Recorded engine patch checksum differs: ${patch_path}")
        endif()
        execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${VULKANA_DIFFUSION_ENGINE_DIR}"
            apply --reverse --check "${patch_file}"
            RESULT_VARIABLE patch_result OUTPUT_QUIET ERROR_QUIET)
        if(NOT patch_result EQUAL 0)
            message(FATAL_ERROR "Required engine patch is missing or differs: ${patch_path}. See src/diffusion/patches/README.md")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${patch_file}")
    endforeach()
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../engine-lock.json")
if(VULKANA_DIFFUSION_RELEASE)
    set(SD_CUDA_CUTLASS_PATH "${VULKANA_DIFFUSION_ENGINE_DIR}/thirdparty/cutlass"
        CACHE PATH "Pinned CUTLASS source" FORCE)
endif()
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_RPC OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
if(SD_CUDA AND SD_VULKAN)
    message(FATAL_ERROR "Build CUDA and Vulkan workers in separate directories")
endif()
add_subdirectory("${VULKANA_DIFFUSION_ENGINE_DIR}" engine EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../profiles.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../profiles.json" profile_json)
configure_file(Profiles.h.in "${CMAKE_CURRENT_BINARY_DIR}/Profiles.h" @ONLY)
add_executable(vulkana-diffusion-worker main.cpp)
target_include_directories(vulkana-diffusion-worker PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "../include")
target_include_directories(vulkana-diffusion-worker PRIVATE
    "${VULKANA_DIFFUSION_ENGINE_DIR}/include" "${VULKANA_DIFFUSION_ENGINE_DIR}/thirdparty")
target_link_libraries(vulkana-diffusion-worker PRIVATE stable-diffusion Threads::Threads)
if(VULKANA_DIFFUSION_RELEASE AND UNIX AND NOT APPLE)
    # The first directory belongs only to this worker. The second permits an
    # enclosing portable app to supply its reviewed GCC/OpenMP runtime.
    # DT_RPATH also resolves cuBLAS's transitive dependencies without rewriting
    # the redistributable NVIDIA libraries or altering the editor environment.
    set_target_properties(vulkana-diffusion-worker PROPERTIES
        BUILD_WITH_INSTALL_RPATH TRUE
        INSTALL_RPATH "$ORIGIN/../lib/vulkana-diffusion/cuda;$ORIGIN/../lib")
    target_link_options(vulkana-diffusion-worker PRIVATE "LINKER:--disable-new-dtags")
endif()
target_compile_definitions(vulkana-diffusion-worker PRIVATE VULKANA_DIFFUSION_ENGINE_COMMIT="${engine_revision}")
if(SD_CUDA)
    target_compile_definitions(vulkana-diffusion-worker PRIVATE VULKANA_DIFFUSION_BACKEND="cuda")
elseif(SD_VULKAN)
    target_compile_definitions(vulkana-diffusion-worker PRIVATE VULKANA_DIFFUSION_BACKEND="vulkan")
else()
    target_compile_definitions(vulkana-diffusion-worker PRIVATE VULKANA_DIFFUSION_BACKEND="cpu")
endif()

        set_tests_properties(vulkana_diffusion_convrot_cuda PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77 LABELS "diffusion;kernel;gpu")
