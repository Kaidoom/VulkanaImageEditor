# This host-side module has no inference-engine, CUDA, GUI or model dependency.
cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(VulkanaDiffusion LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_AUTOMOC ON)
    find_package(Qt6 REQUIRED COMPONENTS Core)
    include(CTest)
    if(BUILD_TESTING)
        find_package(Qt6 REQUIRED COMPONENTS Test)
    endif()
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/profiles.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/profiles.json" profile_json)
configure_file(worker/Profiles.h.in "${CMAKE_CURRENT_BINARY_DIR}/Profiles.h" @ONLY)
add_library(imageeditor_diffusion STATIC
    src/PromptPresets.cpp src/WorkerClient.cpp src/ModelProfiles.cpp
    include/imageeditor/diffusion/WorkerClient.h)
target_include_directories(imageeditor_diffusion PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(imageeditor_diffusion PUBLIC Qt6::Core)
if(COMMAND imageeditor_enable_warnings)
    imageeditor_enable_warnings(imageeditor_diffusion)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND BUILD_TESTING)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../tests/DiffusionTests.cmake")
endif()
